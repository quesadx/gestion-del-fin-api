```typescript
    user.sessionVersion,
    user.isAdmin,

console.log(`Setup complete: ${Object.keys(tokens).length} tokens generated`);
```

```typescript
// tests/e2e/global.teardown.ts
import fs from 'fs';
import path from 'path';

async function globalTeardown(): Promise<void> {
  const tokensFile = path.join(__dirname, '.auth', 'tokens.json');
  if (fs.existsSync(tokensFile)) {
    fs.unlinkSync(tokensFile);
    console.log('Teardown: removed tokens.json');
  }
  // NOTE: Database is NOT truncated in teardown.
  // Setup handles TRUNCATE on the next run. This avoids teardown failures
  // leaving the DB in an unknown state (truncate-on-next-run is safer).
  console.log('E2E teardown complete');
}

export default globalTeardown;
```

**Why no DB truncation in teardown (per PITFALLS.md pitfall 10):**

- Teardown runs after ALL tests. If teardown fails (e.g., DB connection dropped), the database would be left partially truncated — an unknown state.
- Setup runs truncation BEFORE tests. If setup fails during truncation, tests don't run. If setup succeeds, tests get a clean DB.
- This is the "truncate-on-next-run" pattern — safer and simpler than trying to clean up at the end.

Do NOT import `test` or `expect` from `@playwright/test` in this file.
